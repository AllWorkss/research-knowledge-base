Upstox Developer Credentials

upstox_algo_bot.py (Ready to Run Python Script) 
import upstox_client
from upstox_client.rest import ApiException

# 1. Initialize Upstox API Configuration
configuration = upstox_client.Configuration()
configuration.access_token = 'YOUR_DAILY_ACCESS_TOKEN'

api_instance = upstox_client.OrderApi(upstox_client.ApiClient(configuration))

# 2. Define Auto BUY Order Function
def place_algo_buy_order(symbol_key, qty, trigger_price):
    body = upstox_client.PlaceOrderRequest(
        quantity=qty,
        product='I', # 'I' for Intraday, 'D' for Delivery
        validity='DAY',
        price=trigger_price,
        tag='ZENITH_ALGO',
        instrument_token=symbol_key,
        order_type='LIMIT',
        transaction_type='BUY',
        disclosed_quantity=0,
        trigger_price=0.0,
        is_amo=False
    )
    try:
        api_response = api_instance.place_order(body, api_version='2.0')
        print(f"✅ BUY Order Executed! Order ID: {api_response.data.order_id}")
        return api_response.data.order_id
    except ApiException as e:
        print(f"❌ Exception when placing order: {e}")

ICICI Breeze Credentials

icici_breeze_bot.py (Ready to Run Python Script) 
from breeze_connect import BreezeConnect

# 1. Initialize Breeze Connect SDK
breeze = BreezeConnect(api_key="YOUR_APP_KEY")
breeze.generate_session(api_secret="YOUR_SECRET_KEY", session_token="YOUR_SESSION_TOKEN")

# 2. Place Order on ICICI Direct
def place_breeze_buy_order(stock_code, qty, price):
    response = breeze.place_order(
        stock_code=stock_code,       # e.g., 'NIFTY' or 'RELIANCE'
        exchange_code="NSE",
        product="margin",            # 'margin' for intraday, 'cash' for delivery
        action="buy",
        order_type="limit",
        stoploss="0",
        quantity=str(qty),
        price=str(price),
        validity="day"
    )
    print("✅ ICICI Breeze Order Response:", response)
    return response
